        // تأثير طباعة النصوص
        function typeWriter(text, elementId, btnId, speed = 50) {
            let i = 0;
            let elem = document.getElementById(elementId);
            elem.innerHTML = "";
            function type() {
                if (i < text.length) {
                    elem.innerHTML += text.charAt(i);
                    i++;
                    setTimeout(type, speed);
                } else if(btnId) {
                    document.getElementById(btnId).style.display = "inline-block";
                }
            }
            type();
        }

        // التنقل بين الصفحات
        function nextPage(pageNum) {
            document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
            document.getElementById('p' + pageNum).classList.add('active');

            if(pageNum === 2) {
                typeWriter("(بس ماراح اسمح لك تشوفه بهالسهولة لازم تبذل جهد شوي🤣🤣)", "type2", "btn2");
            }
        }

        // تشغيل الصفحة الأولى

        // التحقق من الرقم السري
        let wrongAttempts = 0;
        function checkPass() {
            let val = document.getElementById('pass-input').value;
            if(val === "1404") {
                nextPage(4);
                initCrossword1();
            } else {
                wrongAttempts++;
                if(wrongAttempts >= 2) {
                    document.getElementById('pass-hint').innerText = "تلميح: رمز غرفتنا السرية للعبة القضية 🤫";
                } else {
                    document.getElementById('pass-hint').innerText = "رمز خاطئ، حاول مرة أخرى!";
                }
            }
        }

        // لعبة الكلمات المتقاطعة 1 (بنفسجي)
        let current1 = [];
        function initCrossword1() {
            const letters = ["ب", "ن", "ف", "س", "ج", "ي", "أ", "ر", "م", "و", "ل"];
            let kb = document.getElementById('kb1');
            kb.innerHTML = "";
            letters.sort(() => Math.random() - 0.5).forEach(l => {
                let btn = document.createElement('button');
                btn.className = 'key-btn';
                btn.innerText = l;
                btn.onclick = () => addLetter1(l);
                kb.appendChild(btn);
            });
        }

        function addLetter1(l) {
            if(current1.length < 6) {
                current1.push(l);
                updateGrid('grid1', current1);
                if(current1.length === 6) {
                    if(current1.join('') === "بنفسجي") {
                        setTimeout(() => { nextPage(5); initCrossword2(); }, 500);
                    } else {
                        setTimeout(() => { current1 = []; updateGrid('grid1', current1); }, 600);
                    }
                }
            }
        }

        // لعبة الكلمات المتقاطعة 2 (فروسي)
        let current2 = [];
        function initCrossword2() {
            const letters = ["ف", "ر", "و", "س", "ي", "م", "ك", "ل", "هـ"];
            let kb = document.getElementById('kb2');
            kb.innerHTML = "";
            letters.sort(() => Math.random() - 0.5).forEach(l => {
                let btn = document.createElement('button');
                btn.className = 'key-btn';
                btn.innerText = l;
                btn.onclick = () => addLetter2(l);
                kb.appendChild(btn);
            });
        }

        function addLetter2(l) {
            if(current2.length < 5) {
                current2.push(l);
                updateGrid('grid2', current2);
                if(current2.length === 5) {
                    if(current2.join('') === "فروسي") {
                        setTimeout(() => { nextPage(6); initPuzzle(); }, 500);
                    } else {
                        setTimeout(() => { current2 = []; updateGrid('grid2', current2); }, 600);
                    }
                }
            }
        }

        function updateGrid(gridId, arr) {
            let boxes = document.getElementById(gridId).children;
            for(let i=0; i<boxes.length; i++) {
                boxes[i].innerText = arr[i] || "";
            }
        }

        // لعبة البزل (16 قطعة)
        let puzzleOrder = Array.from({length: 16}, (_, i) => i);
        function initPuzzle() {
            let container = document.getElementById('puzzle-container');
            container.innerHTML = "";
            let shuffled = [...puzzleOrder].sort(() => Math.random() - 0.5);

            shuffled.forEach((pieceIdx) => {
                let div = document.createElement('div');
                div.className = 'puzzle-piece';
                div.dataset.index = pieceIdx;
                
                let row = Math.floor(pieceIdx / 4);
                let col = pieceIdx % 4;
                div.style.backgroundImage = "url('https://i.ibb.co/JNNRf62/5-D4-FA3-F5-08-BC-4-AA8-B6-BA-04-E08881-F757.jpg')";
                div.style.backgroundPosition = `-${col * 50}px -${row * 50}px`;
                
                div.onclick = () => selectPuzzlePiece(div);
                container.appendChild(div);
            });
        }

        let selectedPiece = null;
        function selectPuzzlePiece(piece) {
            if(!selectedPiece) {
                selectedPiece = piece;
                piece.style.border = "2px solid red";
            } else {
                let bgPos = selectedPiece.style.backgroundPosition;
                let idx = selectedPiece.dataset.index;

                selectedPiece.style.backgroundPosition = piece.style.backgroundPosition;
                selectedPiece.dataset.index = piece.dataset.index;
                selectedPiece.style.border = "none";

                piece.style.backgroundPosition = bgPos;
                piece.dataset.index = idx;

                selectedPiece = null;
                checkPuzzleSolved();
            }
        }

        function checkPuzzleSolved() {
            let pieces = document.querySelectorAll('.puzzle-piece');
            let isSolved = true;
            pieces.forEach((p, index) => {
                if(parseInt(p.dataset.index) !== index) isSolved = false;
            });

            if(isSolved) {
                document.getElementById('save-btn').style.display = 'inline-block';
                document.getElementById('puzzle-next').style.display = 'inline-block';
            }
        }

        // العد التنازلي
        function startCountdownPage() {
            document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
            document.getElementById('page-countdown').classList.add('active');

            let count = 4;
            let timerElem = document.getElementById('timer-num');
            
            let interval = setInterval(() => {
                count--;
                if(count > 0) {
                    timerElem.innerText = count;
                } else {
                    clearInterval(interval);
                    showFinalPage();
                }
            }, 1000);
        }

        // الصفحة الأخيرة + إطلاق الاحتفالات المتطايرة

            document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));

            });

                    origin: { x: 0 }
                });
                confetti({
                    particleCount: 50,
                    angle: 120,
                    spread: 55,
                    origin: { x: 1 }
                });
            }, 3000);
        }
